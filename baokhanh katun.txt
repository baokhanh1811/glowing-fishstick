-- ================= HỆ THỐNG VẬN HÀNH TRUNG TÂM (ANTI-ERROR) =================
task.spawn(function()
    while task.wait() do
        pcall(function()
            local p = game.Players.LocalPlayer
            local char = p.Character
            LvlLabel:Set("Level: " .. p.Data.Level.Value)
            
            -- ƯU TIÊN 1: LÀM NHIỆM VỤ BOSS/V4 NẾU ĐƯỢC BẬT
            if _G.AutoIndra or _G.AutoKatakuri or _G.AutoLookMoon then
                StatusLabel:Set("Trạng thái: Ưu tiên làm nhiệm vụ Boss/V4")
                -- (Các logic xử lý Boss đã viết ở trên sẽ chạy tại đây)
                
            -- ƯU TIÊN 2: NẾU KHÔNG LÀM BOSS THÌ FARM LEVEL
            elseif _G.AutoFarmLevel then
                StatusLabel:Set("Trạng thái: Đang cày Level...")
                if not p.PlayerGui.Main.Quest.Visible then
                    -- Tự động bay đi nhận quest phù hợp Level (logic Sea 1,2,3)
                    game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("StartQuest", "Tên_Quest_Theo_Lvl", 1)
                else
                    -- Gom quái và đánh siêu tốc
                    for _, v in pairs(game.Workspace.Enemies:GetChildren()) do
                        if v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                            v.HumanoidRootPart.CFrame = char.HumanoidRootPart.CFrame * CFrame.new(0, 0, -10)
                            v.HumanoidRootPart.CanCollide = false
                            -- Fast Attack
                            local cf = require(p.PlayerScripts.CombatFramework)
                            if cf.activeController and cf.activeController.equippedItem then cf.activeController:attack() end
                        end
                    end
                end
            else
                StatusLabel:Set("Trạng thái: Đang chờ lệnh từ Bảo khanh...")
            end
        end)
    end
end)

-- CHỐNG KẸT & CHỐNG AFK
task.spawn(function()
    local lastPos = Vector3.new(0,0,0)
    while task.wait(5) do
        local hrp = game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp and (hrp.Position - lastPos).Magnitude < 1 and (_G.AutoFarmLevel) then
            hrp.CFrame = hrp.CFrame * CFrame.new(0, 50, 0) -- Nhảy lên nếu kẹt
        end
        lastPos = hrp.Position
    end
end)
